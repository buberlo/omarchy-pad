# PulseAudio for the iSH desktop: one sink that streams raw PCM to the iOS app.
# The iOS side (app/Desktop/ISHAudioBridge.swift) reads /tmp/ishaudio/pcm, which is a
# host FIFO under the fakefs data directory.
.fail
load-module module-native-protocol-unix auth-anonymous=1 socket=/tmp/ishaudio/native

# use_system_clock_for_timing: the sink paces itself and drops data when nobody reads,
# so a missing or stalled reader can never block guest audio clients.
load-module module-pipe-sink file=/tmp/ishaudio/pcm sink_name=ishaudio format=s16le rate=48000 channels=2 use_system_clock_for_timing=yes sink_properties=device.description=iPad
set-default-sink ishaudio

# Stops the sink (and so the FIFO traffic) a few seconds after the last stream ends;
# the host gives the iOS audio session back to other apps once data stops.
load-module module-suspend-on-idle timeout=3

# The iPad microphone. The iOS side (app/Desktop/ISHMicBridge.swift) writes s16le mono
# 48 kHz into /tmp/ishaudio/mic. The source only reads while a client records, so the
# host sees the FIFO drain and starts the microphone then, and stops it when data piles up.
load-module module-pipe-source file=/tmp/ishaudio/mic source_name=ipad_mic format=s16le rate=48000 channels=1 source_properties=device.description="iPad\ Microphone"
set-default-source ipad_mic

.nofail
load-module module-stream-restore restore_device=false
load-module module-position-event-sounds
