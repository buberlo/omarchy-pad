=== how the desktop starts the X bridge ===
#!/bin/sh
# Runs an X11-only app (VLC 3's Qt interface, Dillo, ...) inside its own rootful
# Xwayland, which shows up through ishwl as one ordinary window.
#   ishwl-x11 [-g WIDTHxHEIGHT] command [args...]
# Rootless Xwayland would need an X window manager inside ishwl; one X server per app
# keeps ishwl a plain Wayland compositor. ishwl-x11-wm makes the app's main window
# fill the server's screen. The server, and with it the WM, exits with the app.
geometry=1024x700
if [ "$1" = "-g" ]; then
    geometry=$2
    shift 2
fi
[ $# -gt 0 ] || { echo "usage: ishwl-x11 [-g WxH] command [args...]" >&2; exit 2; }

display=1
while [ -e "/tmp/.X11-unix/X$display" ] || [ -e "/tmp/.X$display-lock" ]; do
    display=$((display + 1))
done
